cmake_minimum_required(VERSION 3.22)
project(hannah LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "")
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "")
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
if(ANDROID_ABI STREQUAL "x86_64")
    foreach(feature AVX AVX2 AVX512 FMA F16C BMI2 SSE42)
        set(GGML_${feature} OFF CACHE BOOL "")
    endforeach()
endif()
set(GGML_OPENMP OFF CACHE BOOL "")
set(GGML_LLAMAFILE OFF CACHE BOOL "")
set(GGML_BACKEND_DL OFF CACHE BOOL "")
FetchContent_Declare(llamacpp URL https://github.com/ggml-org/llama.cpp/archive/refs/tags/b5046.tar.gz)
FetchContent_MakeAvailable(llamacpp)
add_library(hannah SHARED hannah.cpp)
target_compile_features(hannah PRIVATE cxx_std_17)
target_link_libraries(hannah PRIVATE llama android log)
